  function showStatus(message, type = "info") {
    const statusEl = document.getElementById("emailStatus"); // Reuse this if you don't want a separate element
    if (!statusEl) return;

    statusEl.textContent = message;
    statusEl.style.display = "block";
    statusEl.style.color = {

      error: "#d75a4a",
      info: "#555",
    }[type] || "#555";

    statusEl.style.fontWeight = "500";

    if (window._pdfStatusTimeout) clearTimeout(window._pdfStatusTimeout);
    if (type === "info") {
      window._pdfStatusTimeout = setTimeout(() => {
        statusEl.style.display = "none";
        statusEl.textContent = "";

  function wirePrint(buttonSelector, { root, header=null, title="Export" }){
    const btn = document.querySelector(buttonSelector);
    if (!btn) return;

    btn.addEventListener("click", () => {
      const rootEl   = document.querySelector(root);
      const headerEl = header ? document.querySelector(header) : null;
      if (!rootEl){ alert("Nothing to export."); return; }

      // build wrapper
      const wrap = document.createElement("div");
      wrap.id = "aa-print";
      if (headerEl && !rootEl.contains(headerEl)) wrap.appendChild(headerEl.cloneNode(true));
      wrap.appendChild(rootEl.cloneNode(true));
      wrap.querySelectorAll('.sr-loading, #sliproot-loading, [data-aa-pdf="exclude"], .no-print, script, link[rel="preload"]').forEach(n=>n.remove());

      // inject PRINT CSS only for this print session
      const style = document.createElement("style");
      style.id = "aa-print-temp-style";
      style.setAttribute("media","print");
      style.textContent = PRINT_CSS;

      // lock out multi-taps
      btn.disabled = true; btn.classList.add("pulsing");
      document.head.appendChild(style);
      document.body.appendChild(wrap);
      document.body.classList.add("aa-printing");

      const cleanup = () => {
        document.body.classList.remove("aa-printing");
        wrap.remove();
        style.remove();            // <-- remove print CSS so it can’t affect screen
        btn.disabled = false; btn.classList.remove("pulsing");
        window.removeEventListener("afterprint", cleanup);
      };
      window.addEventListener("afterprint", cleanup, { once: true });

      try { window.print(); }
      finally { setTimeout(cleanup, 1200); } // iOS/Safari fallback
    });
  }


  // Wire each product

  wirePrint("#dl-ad-pdf",       { root: ".markdown-plan, #markdown-plan", header: ".align-header", title: "AlignDaily_Plan" });
  wirePrint("#dl-as-pdf",       { root: ".report-container", title: "Alignment_Guide" });
})();
