<script>
document.addEventListener("DOMContentLoaded", () => {
  const planOutput = document.getElementById("plan-output");
  const product = "aligndaily";

  // ---------- Render plan ----------
  try {
    const rawPlan = localStorage.getItem("aligndailyPlan");
    if (!planOutput) throw new Error("plan-output not found");
    if (!rawPlan || rawPlan.trim().length < 100) {
      planOutput.innerHTML = `<p style="text-align:center;color:#999;">No plan found. Please return to the <a href="/aligndaily">form</a> to generate a new one.</p>`;
    } else {
      planOutput.innerHTML = `<div class="markdown-plan">${rawPlan}</div>`;
// Normalize list markers inside the plan (hide by default)
const planRoot = document.querySelector(".markdown-plan");
if (planRoot) {
  planRoot.querySelectorAll("ul:not(.with-bullets), ol:not(.with-bullets)").forEach(list => {
    list.style.listStyle = "none";
    list.style.paddingLeft = "0";
    list.style.marginLeft  = "0";
    list.querySelectorAll("li").forEach(li => {
      // extra belt-and-suspenders for Safari quirks
      li.style.listStyle = "none";
    });
  });
}

    }
  } catch (e) {
    console.error(e);
    planOutput.innerHTML = `<p style="color:#b00020;">Something went wrong displaying your plan.</p>`;
  }

  // ---------- Elements ----------
  const revealBtn   = document.getElementById("revealEmailBtn");
  const inputRow    = document.getElementById("emailInputRow");
  const submitBtn   = document.getElementById("submitEmailBtn");
  const cancelBtn   = document.getElementById("cancelEmailBtn");
  const emailField  = document.getElementById("email");
  const statusMsg   = document.getElementById("emailStatus");
  const pdfBtn      = document.getElementById("dl-ad-pdf");
  const pdfStatusEl = document.getElementById("pdfStatusBadge");

  // ---------- Helpers ----------
  const isEmail = (v) => /^\S+@\S+\.\S+$/.test(String(v||"").trim());
  function setEmailStatus(text="", state=""){
    if (!statusMsg) return;
    statusMsg.textContent = text;
    statusMsg.classList.remove("ok","fail","info");
    if (state) statusMsg.classList.add(state);
  }
  function setPdfStatus(text="", state=""){
    if (!pdfStatusEl) return;
    pdfStatusEl.textContent = text;
    pdfStatusEl.style.display = text ? "inline-block" : "none";
    pdfStatusEl.classList.remove("ok","fail","info");
    if (state) pdfStatusEl.classList.add(state);
  }

  // ---------- Open/Close Email Row (button toggles; Esc & Cancel close) ----------
  let emailOpen = false;
  function openEmailRow(){
    emailOpen = true;
    inputRow.hidden = false;
    revealBtn.setAttribute("aria-expanded","true");
    inputRow.scrollIntoView({ behavior:"smooth", block:"center" });
    setTimeout(() => emailField?.focus({ preventScroll:true }), 200);
  }
  function closeEmailRow(clear=false){
    emailOpen = false;
    inputRow.hidden = true;
    revealBtn.setAttribute("aria-expanded","false");
    if (clear){ if (emailField) emailField.value = ""; setEmailStatus("", ""); }
  }

  revealBtn?.addEventListener("click", () => { emailOpen ? closeEmailRow() : openEmailRow(); });
  cancelBtn?.addEventListener("click", () => closeEmailRow(true));
  document.addEventListener("keydown", e => { if (e.key === "Escape" && emailOpen) closeEmailRow(); });

  // ---------- Submit Email ----------
  submitBtn?.addEventListener("click", async () => {
    const email = (emailField?.value || "").trim();
    const planHTML = document.querySelector(".markdown-plan")?.outerHTML || "";
    if (!planHTML) { setEmailStatus("Plan content missing.", "fail"); return; }
    if (!isEmail(email)) { setEmailStatus("Please enter a valid email address.", "fail"); return; }

    const prevLabel = submitBtn.textContent;
    submitBtn.disabled = true; submitBtn.textContent = "Sending…";
    setEmailStatus("Sending email…", "info");

    try {
      const res = await fetch("https://redacted-pipedream-endpoint.invalid", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, product, html: planHTML })
      });

      const ct = (res.headers.get("content-type") || "").toLowerCase();
      const result = ct.includes("application/json") ? await res.json()
                  : { success: res.ok, raw: await res.text() };

      if (!result.success) throw new Error(result.raw || "Unknown error");

      closeEmailRow(true);
      setEmailStatus("✓ Plan sent to your email.", "ok");
    } catch (err) {
      console.error("[AlignDaily] Email error:", err);
      setEmailStatus("❌ Failed to send. Please try again.", "fail");
    } finally {
      submitBtn.disabled = false;
      submitBtn.textContent = prevLabel || "Submit Email";
    }
  });

  // ---------- (Optional) PDF handler – keep if you re-enable ----------

});

/**
 * Wire a print/export button.
 * @param {string} buttonSelector - CSS selector for the print button
 * @param {Object} cfg - configuration
 * @param {string} cfg.root - selector for main content to export
 * @param {string|null} cfg.header - optional selector for a header to prepend
 * @param {string} cfg.title - logical export title (unused by window.print but reserved)
 * @param {string|null} cfg.extras - optional selector for extra nodes to append
 */
function wirePrint(buttonSelector, cfg){
  const btn = document.querySelector(buttonSelector);
  if (!btn) return;

  const {
    root,
    header = null,
    title = "Export",
    extras = null
  } = cfg || {};

  btn.addEventListener("click", () => {
    const rootEl   = document.querySelector(root);
    const headerEl = header ? document.querySelector(header) : null;
    if (!rootEl){ alert("Nothing to export."); return; }

    // build wrapper
    const wrap = document.createElement("div");
    wrap.id = "aa-print";

    // header (optional)
    if (headerEl && !rootEl.contains(headerEl)) {
      wrap.appendChild(headerEl.cloneNode(true));
    }

    // main content
    wrap.appendChild(rootEl.cloneNode(true));

    // extras (e.g., Flowproofing + Disclaimer that live outside .markdown-plan)
    if (extras) {
      const extraNodes = document.querySelectorAll(extras);
      extraNodes.forEach(sec => {
        // Only append if the original root does *not* already contain it
        if (!rootEl.contains(sec)) {
          wrap.appendChild(sec.cloneNode(true));
        }
      });
    }

    // clean up things we never want in print
    wrap.querySelectorAll(
      '.sr-loading, #sliproot-loading, [data-aa-pdf="exclude"], .no-print, script, link[rel="preload"]'
    ).forEach(n => n.remove());

    // inject PRINT CSS only for this print session
    const style = document.createElement("style");
    style.id = "aa-print-temp-style";
    style.setAttribute("media","print");
    style.textContent = PRINT_CSS;

    // lock out multi-taps
    btn.disabled = true;
    btn.classList.add("pulsing");
    document.head.appendChild(style);
    document.body.appendChild(wrap);
    document.body.classList.add("aa-printing");

    const cleanup = () => {
      document.body.classList.remove("aa-printing");
      wrap.remove();
      style.remove();            // remove print CSS so it can’t affect screen
      btn.disabled = false;
      btn.classList.remove("pulsing");
      window.removeEventListener("afterprint", cleanup);
    };
    window.addEventListener("afterprint", cleanup, { once: true });

    try { window.print(); }
    finally { setTimeout(cleanup, 1200); } // iOS/Safari fallback
  });
}
