    if (!wakeLock) return;

      await wakeLock.release();

      console.log('[AlignmentScan] Error releasing wake lock:', err.name, err.message);

      const r = await fetch(ENTITLEMENTS_URL, { headers: { Authorization: "Bearer " + tok }});
      if (!r.ok) throw 0;
      const e = await r.json();
      const credits = (e.credits && e.credits[PRODUCT]) || 0;

      if (credits > 0) {
        if (submit) submit.disabled = false;
        if (next)   next.disabled   = false; // ✅ enable Next when credits exist
        note.textContent = `You have ${credits} run${credits>1?"s":""} remaining.`;
      } else {
        if (submit) submit.disabled = true;
        if (next)   next.disabled   = true;  // 🔒 keep Next disabled
        note.innerHTML = `No credits left. <a href="/${PRODUCT}">Buy another run</a>.`;
      }
    } catch {
      if (submit) submit.disabled = true;
      if (next)   next.disabled   = true;    // 🔒 keep Next disabled on error
      note.textContent = "Could not verify access.";
    }
  })();

  let currentStep = 1;
  let allowProgressNav = false; // 🔒 Progress bar starts locked
  const totalSteps = steps.length;

  // Step nav: Clickable progress bar after review
  document.querySelectorAll("#progressSteps .step").forEach((stepEl, idx) => {
    stepEl.addEventListener("click", () => {
      if (allowProgressNav && idx + 1 < totalSteps) {
        console.log("🟦 Jumping to step", idx + 1);
        currentStep = idx + 1;
        showStep(currentStep);
        window.scrollTo({ top: 0, behavior: "smooth" });
      }
    });
  });

  function showStep(step) {
    console.log("🔍 Showing step:", step);
    steps.forEach((s) => s.style.display = "none");
    const current = document.querySelector(`.form-step[data-step="${step}"]`);
    if (current) { current.style.display = "block"; console.log("✅ Step found and displayed."); } else { console.warn("❌ Step not found:", step); }

    if (step === totalSteps) {
      allowProgressNav = true;
    }

    prevBtn.style.display = step === 1 ? "none" : "inline-block";
    nextBtn.style.display = step === totalSteps ? "none" : "inline-block";
    submitBtn.style.display = step === totalSteps ? "inline-block" : "none";

    updateProgress(step);
    progressEl?.style.setProperty('--step', step);
  }

  function updateProgress(step) {
    const stepsUI = document.querySelectorAll("#progressSteps .step");
    stepsUI.forEach((el, idx) => {
      if (idx < step) el.classList.add("active");
      else el.classList.remove("active");
    });
  }

  function collectFormData() {
    const data = {};
    const fields = document.querySelectorAll("input, select, textarea");
    const seenRadioGroups = new Set();

    fields.forEach((el) => {
      const name = el.name || el.id;
      if (!name) return;

      if (el.type === "checkbox") {
        if (!data[name]) data[name] = [];
        if (el.checked) data[name].push(el.value);
      } else if (el.type === "radio") {
        if (seenRadioGroups.has(name)) return;
        const selected = document.querySelector(`input[name="${name}"]:checked`);
        data[name] = selected ? selected.value.trim() : null;
        seenRadioGroups.add(name);
      } else if (el.multiple) {
        const selected = Array.from(el.selectedOptions).map((opt) => opt.value);
        data[name] = selected;
      } else if (el.type !== "submit" && el.type !== "button") {
        data[name] = el.value.trim();
      }
    });

    Object.keys(data).forEach((key) => {
      if (typeof data[key] === "string" && data[key].trim() === "") {
        data[key] = null;
      }
    });

    return data;
  }

  form.addEventListener("input", () => {
    const currentData = collectFormData();
    localStorage.setItem("fieldScanDraft", JSON.stringify(currentData));
  });

  // 🔄 Auto-restore from localStorage
  const saved = localStorage.getItem("fieldScanDraft");
  if (saved) {
    const data = JSON.parse(saved);
    Object.entries(data).forEach(([key, value]) => {
      const field = form.elements[key];
      if (!field) return;

      if (Array.isArray(value)) {
        value.forEach(v => {
          const el = form.querySelector(`[name="${key}"][value="${v}"]`);
          if (el) el.checked = true;
        });
      } else {
        field.value = value;
      }
    });
  }

  function renderReview(data) {
  reviewContainer.innerHTML = "";

  steps.forEach((step, i) => {
    const stepNumber = parseInt(step.getAttribute("data-step"), 10);
    if (stepNumber === totalSteps) return;

    const stepTitle = step.querySelector("h2")?.textContent || "";
    const fields = step.querySelectorAll("textarea, input, select");

    const section = document.createElement("div");
    section.className = "review-block";
    section.innerHTML = `<h3 class="review-title">${stepTitle}</h3>`;

    const seenNames = new Set();

    fields.forEach((field) => {
      const key = field.name;
      if (!key || seenNames.has(key)) return;
      seenNames.add(key);

      let label = key;

      if (field.type === "radio" || field.type === "checkbox") {
        // 1) Try legend inside a fieldset
        const fieldset = field.closest("fieldset");
        const legendEl = fieldset?.querySelector("legend");
        if (legendEl) {
          label = legendEl.textContent.trim();
        } else {
          // 2) Try the block's .question-label (your current pattern)
          const block = field.closest(".form-question-block");
          const qLabel = block?.querySelector(".question-label");
          if (qLabel) {
            label = qLabel.textContent.trim();
          } else {
            // 3) Last resort: ARIA-labelledby on group
            const group = field.closest('[role="radiogroup"],[role="group"]');
            const labelledById = group?.getAttribute("aria-labelledby");
            if (labelledById) {
              const labelNode = document.getElementById(labelledById);
              if (labelNode) label = labelNode.textContent.trim();
            }
          }
        }
      } else if (field.id) {
        // textareas, text inputs, etc.
        const labelEl = form.querySelector(`label[for="${field.id}"]`);
        if (labelEl) {
          label = labelEl.textContent.trim();
        }
      }

      const value = data[key];
      if (value !== undefined && value !== "") {
        const item = document.createElement("div");
        item.className = "review-field";
        item.innerHTML = `<strong>${label}</strong><p>${Array.isArray(value) ? value.join(", ") : value}</p>`;

        // click-to-jump back to step
        item.addEventListener("click", () => {
          currentStep = i + 1;
          showStep(currentStep);
          window.scrollTo({ top: 0, behavior: "smooth" });
        });

        section.appendChild(item);
      }
    });

    // clicking anywhere in the block also jumps back
    section.addEventListener("click", () => {
      currentStep = i + 1;
      showStep(currentStep);
      window.scrollTo({ top: 0, behavior: "smooth" });
    });

    reviewContainer.appendChild(section);
  });
}

  function validateCurrentStep() {
    const currentFields = steps[currentStep - 1].querySelectorAll("textarea, input, select");
    let valid = true;
    const seenGroups = new Set();

    currentFields.forEach(field => {
      const name = field.name;
      const type = field.type;

      if ((type === "radio" || type === "checkbox") && !seenGroups.has(name)) {
        const group = steps[currentStep - 1].querySelectorAll(`[name="${name}"]`);
        const oneChecked = Array.from(group).some(input => input.checked);

        if (!oneChecked) {
          valid = false;
          group.forEach(input => input.classList.add("field-error"));
          const fieldset = group[0].closest("fieldset");
          if (fieldset) fieldset.classList.add("fieldset-error");
        } else {
          group.forEach(input => input.classList.remove("field-error"));
          const fieldset = group[0].closest("fieldset");
          if (fieldset) fieldset.classList.remove("fieldset-error");
        }

        seenGroups.add(name);
      }

      if (type !== "radio" && type !== "checkbox") {
        if (!field.checkValidity()) {
          valid = false;
          field.classList.add("field-error");
        } else {
          field.classList.remove("field-error");
        }
      }
    });

    if (!valid) {
      const firstError = steps[currentStep - 1].querySelector(".field-error, .fieldset-error");
      if (firstError) {
        firstError.scrollIntoView({ behavior: "smooth", block: "center" });
      }
    }

    return valid;
  }

  nextBtn.addEventListener("click", () => {
    if (nextBtn.disabled) return; // 🔒 no credits, no advance

    if (currentStep < totalSteps) {
      const isValid = validateCurrentStep();
      if (!isValid) return;

      currentStep++;
      if (currentStep === totalSteps) {
        const data = collectFormData();
        renderReview(data);
      }

      showStep(currentStep);
      statusMsg.textContent = "";
    }
  });

  prevBtn.addEventListener("click", () => {
    if (currentStep > 1) {
      currentStep--;
      showStep(currentStep);
    }
  });

  let loadingInterval;

  function startRotatingMessages(messages, targetElement, interval = 3500) {
    let index = 0;
    if (!targetElement) return;

    targetElement.textContent = messages[index];
    loadingInterval = setInterval(() => {
      index = (index + 1) % messages.length;
      targetElement.textContent = messages[index];
    }, interval);
  }

  function stopRotatingMessages(finalMessage) {
    clearInterval(loadingInterval);
    if (finalMessage && statusMsg) {
      statusMsg.textContent = finalMessage;
    }
  }

  const loadingMessages = [
    "Scanning for hidden contradictions...",
    "Mapping your inner alignment...",
    "Listening for what's underneath...",
    "Locating deeper motivations...",
    "Almost there... still tuning in...",
    "Translating your responses into insight..."
  ];

  // Busy-state helpers
  function setBusy(btn, text = 'Scanning…') {
    if (!btn) return;
    if (!btn.dataset.prev) btn.dataset.prev = btn.textContent;
    btn.setAttribute('aria-busy', 'true');
    btn.textContent = text;
  }
  function clearBusy(btn) {
    if (!btn) return;
    btn.removeAttribute('aria-busy');
    btn.textContent = btn.dataset.prev || 'Start the Scan';
  }

  form.addEventListener("submit", async (e) => {
    e.preventDefault();

    const isValid = validateCurrentStep();
    if (!isValid) return;

    const data = collectFormData();
    console.log("📦 Final collected form data:", data);
    localStorage.setItem("alignmentMapData", JSON.stringify(data));

    // start animation + spinner
    setBusy(submitBtn, 'Scanning…');
    submitBtn.disabled = true;
    startRotatingMessages(loadingMessages, statusMsg);

    try {
      isGenerating = true;           // NEW: mark as in-flight
      await keepScreenAwake();       // 🔒 Keep the screen awake while the guide is generating

      const aa = aaGetToken(); // v1.<data>.<sig> from localStorage/cookie
      const jobId = crypto?.randomUUID?.() ?? String(Date.now());

        const response = await fetch("https://redacted-pipedream-endpoint.invalid", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          ...(aa ? { Authorization: "Bearer " + aa } : {})
        },
        body: JSON.stringify({
          ...data,
          ...(aa ? { aa_session: aa } : {}),
          jobId
        }),
      });

      let result;
      let rawText = "";
      try {
        rawText = await response.text();
        result = rawText ? JSON.parse(rawText) : {};
      } catch {
        result = { raw: rawText };
      }

      // Handle HTTP or explicit backend error
      const apiSuccess = result && result.success === true;
      const apiCode = typeof result?.code === "number" ? result.code : response.status;
      const apiErrorSlug = result?.error || (response.ok ? null : "unexpected_error");

      if (!response.ok || !apiSuccess) {
        const friendly = friendlyScanError(apiErrorSlug, apiCode);
        console.error("Alignment Scan API error:", {
          httpStatus: response.status,
          apiCode,
          apiErrorSlug,
          raw: result,
        });

        stopRotatingMessages(`⚠️ ${friendly}`);
        clearBusy(submitBtn);
        submitBtn.disabled = false;
        return;
      }

      // Success path: extract HTML from the standard shape
      const html =
        result?.guide?.html ??
        result?.html ??
        (typeof result?.raw === "string" ? result.raw : null);

      if (!html || !String(html).trim()) {
        const friendly = friendlyScanError("empty_prompt", apiCode);
        stopRotatingMessages(`⚠️ ${friendly}`);
        clearBusy(submitBtn);
        submitBtn.disabled = false;
        return;
      }

      try {
        localStorage.setItem("alignmentGuide", JSON.stringify({ guide: { html } }));
        localStorage.removeItem("fieldScanDraft");
      } catch {}

      stopRotatingMessages("✅ Success! Redirecting...");
      window.location.href = "/alignment-guide";


     } catch (error) {
      console.error("❌ Submission Error:", error);
      const friendly = friendlyScanError("network_error");
      stopRotatingMessages(`⚠️ ${friendly}`);
      clearBusy(submitBtn);
      submitBtn.disabled = false;
    } finally {
      isGenerating = false;         // NEW: clear flag
      await releaseWakeLock();      // ✅ Always release wake lock
    }
  });

  showStep(currentStep); // 👈 Initial render
});
