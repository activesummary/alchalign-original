  // If we came from a /buy/* page, make “Try again” go back there
  (function () {
    var a = document.getElementById('cancel-try-again');
    if (!a) return;
    try {
      var ref = document.referrer || "";
      var url = new URL(ref, location.origin);
      if (/^\/buy\//.test(url.pathname)) {
        a.href = ref;
      }
    } catch (e) { /* ignore */ }
  })();
</script>
